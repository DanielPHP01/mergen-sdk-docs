    Свой UI поверх сканера¶

    Три уровня свободы — от «перекрасить рамку» до «камера и движок без единого пикселя SDK».

    Уровень Что даёт Инструмент
    Стилизация цвета, рамка, подсказки, прогресс ScannerStyle (см. «Кастомизация вида»)
    Свой оверлей SDK ведёт камеру и движок, весь UI поверх — ваш onState / customOverlay / рендереры
    Headless SDK — только движок и состояние; камера-вью и UI целиком ваши state-handle

    Во всех вариантах координатную работу SDK берёт на себя: cardBoundsInView и scanRoiInView приходят уже в координатах вью (поворот и заполнение экрана учтены) — рисуйте без пересчётов.

    Уровень 2 — свой оверлей¶

    MergenScanView + onState¶

    Самый короткий путь в одностороннем сканировании: передайте onState, и встроенные тексты SDK подавляются, а каждое состояние кадра приходит вам (на главном потоке):

    ZStack {
    MergenScanView(side: .front, license: license,
                   bboxStyle: .none,
                   onState: { state = $0 },
                   onResult: onResult)
    MyOverlay(state: state)     // рамка по state.scanRoiInView, тексты, кнопки
}

    Box {
    MergenScanView(side = MIDSide.FRONT, license = license,
                   bboxStyle = BBoxDrawStyle.NONE,
                   onState = { state = it },
                   onResult = onResult)
    MyOverlay(state)            // рамка по state.scanRoiInView, тексты, кнопки
}

    Свои тексты стройте по state.scanStatus — один типизированный статус на кадр, уже сглаженный SDK (см. «Состояния и события»). Отдельно обработайте Occluded / .occluded: пока он активен, движок не захватывает кадр (палец на данных), и если ваш оверлей об этом промолчит, пользователь увидит просто замерший сканер.

    MergenScannerView с кастомным оверлеем¶

    Полный сканер (двусторонний режим, экран результата, жизненный цикл) с вашим слоем UI:

    MergenScannerView(licenseKey: license, configuration: config, style: style,
                  onFinished: { result in ... },
                  onError:    { error in ... },
                  onEvent:    { event in ... }) { state in
    // ViewBuilder-оверлей: встроенный UI полностью подавлен
    if let bounds = state.cardBoundsInView {
        MyCardFrame(rect: bounds, color: state.activeColor)
    }
}

    let vc = MergenScannerViewController(licenseKey: license, configuration: config)
vc.onFinished = { result in ... }
vc.customOverlayRenderer = { ctx, state, width, height in
    // CGContext-рендерер; state.cardBoundsInView уже в точках вью
}
vc.style = myStyle                  // применяется немедленно
present(vc, animated: true)         // fullScreen
// Дополнительно: delegate (MergenScannerDelegate), events: AsyncStream<ScannerEvent>,
// reset() / startScanning() / stopScanning() / release() / setFlashlightEnabled(_:)

    MergenScannerView(
    licenseJson = license, config = config, style = style,
    showResultScreen = false,
    onFinished = { result -> ... },
    onError    = { e -> ... },
    onEvent    = { event -> ... },
    onClose    = { ... },                       // рисует крестик, если задан
    customOverlay = { state ->                   // BoxScope; встроенный UI подавлен
        state.cardBoundsInView?.let { MyFrame(it, Color(state.activeColorArgb)) }
    }
)

    <com.mergen.sdk.MergenScannerView android:id="@+id/scanner" ... />

    binding.scanner
    .setConfig(config)
    .setStyle(style)
    .setCustomOverlayRenderer { canvas, state, viewWidth, viewHeight ->
        // Canvas-рендерер; state.cardBoundsInView уже в пикселях вью
    }
    .setOnFinishedListener { result -> ... }
    .setOnErrorListener { e -> ... }
    .setOnCloseListener { ... }
    .setOnGalleryClickListener { ... }
    .startScanning(license)

// Также: processGalleryImage(bitmap), setFlashlightEnabled(...), reset(),
// stopScanning() / startScanning(license) для паузы, release() при уходе,
// events: Flow<ScannerEvent>, showResultScreen = false

    Тип рендерера:

    public typealias ScannerUIOverlayRenderer =
    (_ context: CGContext, _ state: ScannerFrameState,
     _ viewWidth: CGFloat, _ viewHeight: CGFloat) -> Void

    fun interface ScannerOverlayRenderer {
    fun onDraw(canvas: Canvas, state: ScannerFrameState, viewWidth: Int, viewHeight: Int)
}

    Уровень 3 — headless¶

    Движок и живое состояние без какого-либо камерного UI от SDK: превью, кнопки, анимации — всё ваше.

    let handle = MergenScannerStateHandle(licenseKey: license, configuration: config)

// Превью: вставьте handle.previewLayer (AVCaptureVideoPreviewLayer) в свою иерархию
handle.updateViewSize(view.bounds.size)     // ОБЯЗАТЕЛЬНО до старта и при ресайзе:
                                            // без него координаты вью не считаются
handle.onFinished = { result in ... }
handle.startScanning()

// Живое состояние: @Published var frameState — подписка через Combine/SwiftUI
// Управление: stopScanning() (пауза, движок тёплый) / release() (освободить камеру)
//             reset() / setFlashlightEnabled(true) / updateStyle(style)
// Галерея без камеры: handle.processGalleryImage(image) → onFinished
// События: handle.events — AsyncStream<ScannerEvent>; каждый доступ создаёт
//          НОВЫЙ стрим — сохраните его один раз

    val handle = rememberMergenScannerState(
    licenseJson = license, config = config,
    onFinished = { result -> ... },
    onError    = { e -> ... },
)

// Превью: свой CameraX PreviewView
AndroidView(factory = { ctx -> PreviewView(ctx).also(handle.bindCamera) })

// Живое состояние: val state by handle.frameState (Compose State)
// Управление: handle.stopScanning() / handle.startScanning() / handle.reset()
//             handle.release() / handle.setFlashlightEnabled(true)
// События: handle.events — Flow<ScannerEvent> (hot: собирайте, пока UI виден)

    Результат headless-скана — MergenIdResult (success, croppedImage, croppedImageBack в двустороннем режиме, confidence, cardVersion, errorCode/errorMessage). Для передачи в быструю сверку соберите из него типизированный результат стороны:

    let scan = MergenSDK.makeScanResult(idResult: idResult, side: .front)

    val scan = Mergen.makeScanResult(idResult, MIDSide.FRONT)

    Фонарик¶

    Ручное управление — setFlashlightEnabled(true/false) на headless-хэндле, XML-вью и UIKit-контроллере. Автоматический режим (autoTorch, включён по умолчанию) зажигает свет после серии тёмных кадров и гаснет при завершении; любой ручной вызов имеет приоритет над автоматикой.
